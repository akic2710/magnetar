function toWorld(cx,cy){
  return { x: clamp((cx-offX)/scale, -80, VW+80), y: clamp((cy-offY)/scale, -80, VH+80) };
}
function onDown(cx,cy){
  var p = toWorld(cx,cy);
  IN.x=p.x; IN.y=p.y; IN.has=true;
  IN.cssX=cx; IN.cssY=cy; IN.clickCssX=cx; IN.clickCssY=cy;
  if(!IN.down){ IN.down=true; IN.justDown=true; }
  IN.clickX=p.x; IN.clickY=p.y; IN.clicked=true;
  boot();
}
function onMove(cx,cy){
  var p=toWorld(cx,cy); IN.x=p.x; IN.y=p.y;
  IN.cssX=cx; IN.cssY=cy;
  if(!IN.has && !IN.isTouch){ IN.has=true; cv.className='reticle'; }
  IN.has=true;
}
function onUp(){ if(IN.down){ IN.down=false; IN.justUp=true; } }

/* Keyboard is a full alternative scheme. Blast is radial, so no aiming is
   needed and arrows / WASD / ZQSD (AZERTY) all work. Escape and Ctrl+W are
   never bound - CrazyGames reserves them. */
var MOVE_KEYS = {
  ArrowLeft:[-1,0], ArrowRight:[1,0], ArrowUp:[0,-1], ArrowDown:[0,1],
  KeyA:[-1,0], KeyD:[1,0], KeyW:[0,-1], KeyS:[0,1],
  KeyQ:[-1,0], KeyZ:[0,-1]
};
window.addEventListener('keydown', function(e){
  if(e.repeat) return;
  IN.keys[e.code]=true; IN.anyKey=true;
  if(e.code==='Space' || e.code==='ShiftLeft' || e.code==='ShiftRight'){ e.preventDefault(); IN.kMag=true; }
  if(MOVE_KEYS[e.code]) e.preventDefault();
  if(e.code==='KeyM'){ A.setMute(!A.muted); A.sUi(); }
  if(e.code==='KeyP'){ togglePause(); }
  if(e.code==='Digit1'||e.code==='Digit2'||e.code==='Digit3'){
    if(G.mode==='levelup') chooseUpgrade(+e.code.slice(5)-1);
  }
  if(e.code==='Enter' && (G.mode==='over' || G.mode==='cores')) restart();
  if(e.code==='KeyC' && G.mode==='over') openCores();
  if(G.mode==='revive' && !G.adBusy){
    if(e.code==='KeyY') takeRevive();
    if(e.code==='KeyN') finishRun();
  }
  boot();
});
window.addEventListener('keyup', function(e){
  IN.keys[e.code]=false;
  if(e.code==='Space' || e.code==='ShiftLeft' || e.code==='ShiftRight'){
    IN.kMag=false;
    if(G.mode==='play'||G.mode==='intro') doBlast();
  }
});
window.addEventListener('blur', function(){
  IN.keys={}; IN.kMag=false; IN.down=false;
  if(G.mode==='play') G.paused=true;
});
document.addEventListener('visibilitychange', function(){
  if(document.hidden){ if(G.mode==='play') G.paused=true; }
});

function keyVector(){
  var vx=0, vy=0;
  for(var k in MOVE_KEYS){ if(IN.keys[k]){ vx+=MOVE_KEYS[k][0]; vy+=MOVE_KEYS[k][1]; } }
  var l=len(vx,vy);
  if(l>0){ vx/=l; vy/=l; }
  return [vx,vy,l>0];
}

function coreById(id){
  for(var i=0;i<CORES.length;i++) if(CORES[i].id===id) return CORES[i];
  return CORES[0];
}
function num(v){ var n = parseInt(v||'0',10); return isNaN(n) ? 0 : n; }
function spentFlux(){
  var l = ownedList(), total = 0;
  for(var i=0;i<CORES.length;i++) if(l.indexOf(CORES[i].id)>=0) total += CORES[i].cost;
  return total;
}
/* Balance is derived, never stored, so it can never disagree with `owned`. */
function getEarned(){ return num(store('mgn.earned')); }
function getFlux(){ return Math.max(0, getEarned() - spentFlux()); }
function addFlux(n){ store('mgn.earned', String(getEarned() + Math.max(0, Math.round(n)))); }

/* Saves written before the earned/spent model stored a balance. */
function migrateSave(){
  if(store('mgn.earned') !== null) return;
  var old = store('mgn.flux');
  if(old === null) return;
  store('mgn.earned', String(num(old) + spentFlux()));
}

/* Merge local and cloud saves without ever taking something away: unlocks
   union, monotonic counters take the max, and the equipped core has to be one
   the player actually owns afterwards. */
function mergeSave(local, cloud){
  var ids = {}, i, l;
  l = (local['mgn.owned']||'prospector').split(',');
  for(i=0;i<l.length;i++) if(l[i]) ids[l[i]] = 1;
  l = (cloud['mgn.owned']||'prospector').split(',');
  for(i=0;i<l.length;i++) if(l[i]) ids[l[i]] = 1;
  var owned = [];
  for(i=0;i<CORES.length;i++) if(ids[CORES[i].id]) owned.push(CORES[i].id);
  if(owned.indexOf('prospector') < 0) owned.unshift('prospector');

  var core = cloud['mgn.core'];
  if(!core || owned.indexOf(core) < 0) core = local['mgn.core'];
  if(!core || owned.indexOf(core) < 0) core = 'prospector';

  return {
    'mgn.owned':    owned.join(','),
    'mgn.core':     core,
    'mgn.earned':   String(Math.max(num(local['mgn.earned']),   num(cloud['mgn.earned']))),
    'mgn.best':     String(Math.max(num(local['mgn.best']),     num(cloud['mgn.best']))),
    'mgn.bestwave': String(Math.max(num(local['mgn.bestwave']), num(cloud['mgn.bestwave']))),
    'mgn.played':   (local['mgn.played']==='1' || cloud['mgn.played']==='1') ? '1' : null
  };
}

/* Switch store() onto the SDK data module, folding in whatever this device
   already had. The SDK migrates its own guest data on sign-in, but guest
   progress written straight to localStorage before the SDK loaded is invisible
   to it - this is what keeps that from being lost. */
function adoptCloudSave(dataModule){
  var local = {}, cloud = {}, i, k;
  for(i=0;i<SYNC_KEYS.length;i++){ k=SYNC_KEYS[i]; local[k] = store(k); }
  saveBackend = dataModule;
  saveCache = {};
  for(i=0;i<SYNC_KEYS.length;i++){ k=SYNC_KEYS[i]; cloud[k] = store(k); }
  var merged = mergeSave(local, cloud);
  for(k in merged) if(merged[k] !== null) store(k, merged[k]);
  return merged;
}

/* The account swaps underneath us on sign-in, so the cache has to go. */
function refreshCloudSave(){ saveCache = {}; }
function ownedList(){ return (store('mgn.owned')||'prospector').split(','); }
function owns(id){ return id==='prospector' || ownedList().indexOf(id)>=0; }
function selectedCore(){
  var id = store('mgn.core') || 'prospector';
  return owns(id) ? id : 'prospector';
}
function equipCore(id){ if(owns(id)) store('mgn.core', id); }
function buyCore(id){
  var c = coreById(id);
  if(!c || owns(id)) return false;
  if(getFlux() < c.cost) return false;
  var l = ownedList();   // the spend is implied by ownership, never deducted
  if(l.indexOf(id) < 0) l.push(id);
  store('mgn.owned', l.join(','));
  return true;
}
/* Depth is worth more than score, so a deep careful run beats farming wave 3.
   The flat term keeps a short first run from feeling like it paid nothing. */
function fluxFor(score, wave){ return 15 + Math.floor(score/150) + wave*4; }

/* ------------------------------- spawning ------------------------------ */
function addScrap(anywhere){
  var x,y,tries=0;
  do{
    x = rnd(40, VW-40); y = rnd(40, VH-40); tries++;
  } while(tries<12 && len(x-P.x, y-P.y) < (anywhere?120:220));
  scrap.push({ x:x, y:y, vx:rnd(-16,16), vy:rnd(-16,16), r:rnd(5,8), a:rnd(0,TAU), sp:rnd(-2,2), life:0 });
}

function edgePoint(){
  var s = ri(0,3), m = 46;
  if(s===0) return {x:rnd(m,VW-m), y:m};
  if(s===1) return {x:VW-m, y:rnd(m,VH-m)};
  if(s===2) return {x:rnd(m,VW-m), y:VH-m};
  return {x:m, y:rnd(m,VH-m)};
}

function makeEnemy(type, x, y){
  var d = ETYPE[type];
  var wave = Math.max(1,G.wave);
  var hp = d.hp;
  if(type==='boss') hp = 30 + Math.floor(wave/5-1)*20;
  else if(wave>8) hp += Math.floor((wave-8)/3);
  return {
    type:type, x:x, y:y, vx:0, vy:0, r:d.r,
    hp:hp, maxHp:hp, spd:d.spd*(1+Math.min(0.80, wave*0.030)),
    col:d.col, xp:d.xp, sc:d.sc,
    t:rnd(0,10), fire:rnd(0.8,2.2), flash:0, dead:false, charge:0,
    phase:0, phaseT:0, wob:rnd(0,TAU), spawnA:0.6
  };
}

function spawnAt(type, pt){
  var e = makeEnemy(type, pt.x, pt.y);
  enemies.push(e);
  waves.push({x:pt.x, y:pt.y, r:6, max:type==='boss'?110:46, life:0.6, col:e.col, w:3});
  return e;
}

function startWave(n){
  G.wave = n;
  G.waveT = 0;
  G.healedThisWave = false;
  // Wave 30 is a deep run; report progress against that as "complete".
  if(window.CG) window.CG.progress(Math.min(100, n/30*100));
  G.spawnQ.length = 0;
  var isBoss = (n % 5 === 0);
  if(isBoss){
    G.waveBannerTxt = 'WAVE ' + n + '  •  GUARDIAN';
    G.spawnQ.push({t:'boss', at:0.35, pt:{x:VW/2, y:120}});
    A.sBoss();
  } else {
    G.waveBannerTxt = 'WAVE ' + n;
    // Gentle first waves (first impression decides bounce rate on a portal),
    // saturating through the mid game, then steady linear growth so late
    // waves keep escalating instead of plateauing.
    var budget = Math.round(3 + 30*(1 - Math.exp(-n/10)) + Math.max(0, n-12)*2.0);

    // Weighted pool: drones thin out as tougher types take over the budget,
    // so wave 25 is not just fifty drones.
    var pool = [{t:'drone', c:1, w:Math.max(1.2, 8 - n*0.32)}];
    if(n>=2) pool.push({t:'spinner',  c:2, w:2.6});
    if(n>=3) pool.push({t:'shooter',  c:3, w:1.8 + n*0.14});
    if(n>=6) pool.push({t:'splitter', c:4, w:1.2 + n*0.12});
    if(n>=8) pool.push({t:'brute',    c:7, w:0.9 + n*0.10});
    if(n>=11) pool.push({t:'lancer',  c:5, w:1.4 + n*0.13});

    var list=[], guard=0;
    while(budget>0 && guard++<160){
      var opts=[], total=0, i;
      for(i=0;i<pool.length;i++) if(pool[i].c<=budget){ opts.push(pool[i]); total+=pool[i].w; }
      if(!opts.length) break;
      var roll = Math.random()*total, p = opts[opts.length-1];
      for(i=0;i<opts.length;i++){ roll -= opts[i].w; if(roll<=0){ p = opts[i]; break; } }
      budget -= p.c; list.push(p.t);
    }
    for(var j=0;j<list.length;j++) G.spawnQ.push({t:list[j], at:j*0.17, pt:edgePoint()});
    A.sWave();
  }
  G.waveBanner = 2.0;
}
